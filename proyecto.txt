// Estados permitidos para una reserva
const ESTADOS = ['pendiente', 'confirmada', 'atendida', 'cancelada'];

// Toma solo los campos permitidos del cuerpo y les quita espacios
function limpiar(cuerpo) {
  const datos = {};
  ['clienteNombre', 'telefono', 'servicio', 'fechaHora', 'estado'].forEach((campo) => {
    if (typeof cuerpo[campo] === 'string') datos[campo] = cuerpo[campo].trim();
  });
  return datos;
}

// Devuelve un mensaje de error o null; completo = true exige los obligatorios (crear)
function validar(datos, completo) {
  if (completo) {
    for (const campo of ['clienteNombre', 'servicio', 'fechaHora']) {
      if (!datos[campo]) return 'Falta el campo obligatorio: ' + campo;
    }
  }
  for (const campo of ['clienteNombre', 'servicio', 'fechaHora']) {
    if (campo in datos && !datos[campo]) return 'El campo ' + campo + ' no puede estar vacÃ­o';
  }
  if (datos.estado && !ESTADOS.includes(datos.estado)) {
    return 'Estado no vÃ¡lido. Usa: ' + ESTADOS.join(', ');
  }
  return null;
}

// InyecciÃ³n de dependencias: recibe el servicio de la base y el notificador desde afuera
function crearReservaControlador(servicio, notificador) {
  return {
    // POST /reservas: guarda en la base y luego envÃ­a a la API externa
    crear: async (req, res) => {
      try {
        const datos = limpiar(req.body || {});
        const error = validar(datos, true);
        if (error) return res.status(400).json({ error });
        // Paso 1: guardo en la base de datos
        const reserva = await servicio.crear(datos);
        // Paso 2: envÃ­o a la API externa; si falla, la reserva igual queda guardada
        let notificacion = 'no enviada';
        try {
          const externo = await notificador.notificar(reserva);
          notificacion = 'enviada (id externo ' + externo.id + ')';
        } catch (e) {
          console.error('No se pudo notificar a la API externa:', e.message);
        }
        res.status(201).json({ ...reserva, notificacion });
      } catch (error) {
        console.error('Error en POST /reservas:', error.message);
        res.status(500).json({ error: 'No se pudo crear la reserva' });
      }
    },

    // GET /reservas: todas
    listar: async (req, res) => {
      try { res.json(await servicio.listar()); }
      catch (error) {
        console.error('Error en GET /reservas:', error.message);
        res.status(500).json({ error: 'No se pudieron obtener las reservas' });
      }
    },

    // GET /reservas/:id: una por id (404 si no existe)
    obtener: async (req, res) => {
      try {
        const id = Number(req.params.id);
        if (!Number.isInteger(id)) return res.status(400).json({ error: 'El id debe ser un nÃºmero' });
        const reserva = await servicio.obtener(id);
        if (!reserva) return res.status(404).json({ error: 'Reserva no encontrada' });
        res.json(reserva);
      } catch (error) {
        console.error('Error en GET /reservas/:id:', error.message);
        res.status(500).json({ error: 'No se pudo obtener la reserva' });
      }
    },

    // PUT /reservas/:id: modifica (404 si no existe)
    actualizar: async (req, res) => {
      try {
        const id = Number(req.params.id);
        if (!Number.isInteger(id)) return res.status(400).json({ error: 'El id debe ser un nÃºmero' });
        const datos = limpiar(req.body || {});
        const error = validar(datos, false);
        if (error) return res.status(400).json({ error });
        const reserva = await servicio.actualizar(id, datos);
        if (!reserva) return res.status(404).json({ error: 'Reserva no encontrada' });
        res.json(reserva);
      } catch (error) {
        console.error('Error en PUT /reservas/:id:', error.message);
        res.status(500).json({ error: 'No se pudo actualizar la reserva' });
      }
    },

    // DELETE /reservas/:id: elimina y confirma (404 si no existe)
    eliminar: async (req, res) => {
      try {
        const id = Number(req.params.id);
        if (!Number.isInteger(id)) return res.status(400).json({ error: 'El id debe ser un nÃºmero' });
        const reserva = await servicio.eliminar(id);
        if (!reserva) return res.status(404).json({ error: 'Reserva no encontrada' });
        res.json({ mensaje: 'Reserva eliminada', id });
      } catch (error) {
        console.error('Error en DELETE /reservas/:id:', error.message);
        res.status(500).json({ error: 'No se pudo eliminar la reserva' });
      }
    }
  };
}

module.exports = crearReservaControlador;

// Cargo las variables del archivo .env (ahÃ­ estÃ¡ mi clave de Cal.com, fuera del cÃ³digo)
require('dotenv').config();
// Importo axios, la librerÃ­a que me deja hacer peticiones a internet
const axios = require('axios');

// DirecciÃ³n base de JSONPlaceholder, la API de datos de prueba que uso para los clientes
const URL_BASE = 'https://jsonplaceholder.typicode.com';
// DirecciÃ³n base de la API de Cal.com, que uso para los servicios y las citas
const URL_CAL = 'https://api.cal.com/v2';

// FunciÃ³n que arma los encabezados que Cal.com exige; recibe la versiÃ³n de la API a usar
function encabezadosCal(version) {
  // Devuelvo un objeto con los dos encabezados
  return {
    // Authorization: aquÃ­ va mi clave. "Bearer" es la forma en que Cal.com espera recibirla,
    // y process.env.CAL_API_KEY es el valor guardado en el archivo .env
    Authorization: 'Bearer ' + process.env.CAL_API_KEY,
    // cal-api-version: le digo a Cal.com quÃ© versiÃ³n de su API quiero usar
    'cal-api-version': version
  };
}

// FunciÃ³n que trae la lista de clientes de prueba
// async porque pedir datos por internet tarda y la funciÃ³n tiene que esperar
async function obtenerClientes() {
  // await = espera a que JSONPlaceholder responda; pido /users y no mando clave porque no la pide
  const respuesta = await axios.get(`${URL_BASE}/users`);
  // respuesta.data es el JSON con los datos; eso es lo que devuelvo
  return respuesta.data;
}

// FunciÃ³n que trae los servicios de la barberÃ­a (los tipos de evento de Cal.com)
async function obtenerServicios() {
  // Pido /event-types a Cal.com y le mando los encabezados con mi clave y la versiÃ³n 2024-06-14
  const respuesta = await axios.get(`${URL_CAL}/event-types`, { headers: encabezadosCal('2024-06-14') });
  // Cal.com mete la lista dentro de un campo llamado "data"; la guardo en esta variable
  const datos = respuesta.data.data;
  // Si lo que llegÃ³ es una lista la devuelvo; si no, devuelvo una lista vacÃ­a para no romper la pÃ¡gina
  return Array.isArray(datos) ? datos : [];
}

// FunciÃ³n que trae las citas apartadas (las reservas de Cal.com)
async function obtenerCitas() {
  // Pido /bookings a Cal.com con mi clave; esta ruta pide la versiÃ³n 2026-05-01
  const respuesta = await axios.get(`${URL_CAL}/bookings`, { headers: encabezadosCal('2026-05-01') });
  // La lista de reservas tambiÃ©n viene dentro del campo "data"
  const datos = respuesta.data.data;
  // Devuelvo la lista, o una vacÃ­a si no llegÃ³ una lista
  return Array.isArray(datos) ? datos : [];
}

// Exporto las tres funciones para usarlas en otros archivos (el controlador y las pruebas)
module.exports = { obtenerClientes, obtenerServicios, obtenerCitas };

// ConexiÃ³n a la base de datos y descripciÃ³n de la tabla citas
const AppDataSource = require('../database/data-source');
const Cita = require('../entities/Cita');

// Repositorio: la herramienta del ORM para leer y guardar filas
function repo() { return AppDataSource.getRepository(Cita); }

// READ (todos)
async function listar() { return repo().find({ order: { id: 'ASC' } }); }

// READ (uno por id); devuelve null si no existe
async function obtener(id) { return repo().findOneBy({ id }); }

// CREATE: arma la fila y la guarda; devuelve la fila con su id
async function crear(datos) { return repo().save(repo().create(datos)); }

// UPDATE: busca, cambia solo lo recibido y guarda; null si no existe
async function actualizar(id, datos) {
  const cita = await repo().findOneBy({ id });
  if (!cita) return null;
  repo().merge(cita, datos);
  return repo().save(cita);
}

// DELETE: busca y borra; null si no existe
async function eliminar(id) {
  const cita = await repo().findOneBy({ id });
  if (!cita) return null;
  const copia = { ...cita };
  await repo().remove(cita);
  return copia;
}

module.exports = { listar, obtener, crear, actualizar, eliminar };
