// Protección: solo deja pasar si llega la clave del barbero (guardada en el .env, no en el código)
function soloBarbero(req, res, next) {
  const clave = req.get('x-clave-barbero');
  if (!process.env.CLAVE_BARBERO || clave !== process.env.CLAVE_BARBERO) {
    return res.status(401).json({ error: 'Solo el barbero puede hacer esto' });
  }
  next();
}

    return res.json({ ok: true });
  }
  res.status(401).json({ error: 'Clave incorrecta' });
});

// Toma solo los campos permitidos de un cliente y les quita los espacios
function limpiarCliente(cuerpo) {
  const datos = {};
  ['nombre', 'telefono'].forEach((campo) => {
    if (typeof cuerpo[campo] === 'string') datos[campo] = cuerpo[campo].trim();
  });
  return datos;
}

// Devuelve un mensaje de error, o null si todo está bien
// completo = true exige el nombre (se usa al crear)
function validarCliente(datos, completo) {
  if (completo && !datos.nombre) return 'Falta el campo obligatorio: nombre';
  if ('nombre' in datos && (datos.nombre.length < 2 || datos.nombre.length > 60)) {
    return 'El nombre debe tener entre 2 y 60 caracteres';
  }
  if (datos.telefono && !/^[0-9+().\-\sxX]{7,25}$/.test(datos.telefono)) {
    return 'El teléfono no es válido (solo números, de 7 a 25 caracteres)';
  }
  return null;
}

// Importo winston, la librería que me deja escribir logs con niveles (info, error)
const winston = require('winston');
// Importo fs para poder crear la carpeta donde se guardan los logs
const fs = require('fs');

// Si la carpeta logs no existe, la creo
if (!fs.existsSync('logs')) fs.mkdirSync('logs');

// Creo el logger con su configuración
const logger = winston.createLogger({
  // Nivel mínimo que se registra: info y todo lo más grave (como error)
  level: 'info',
  // Formato de cada línea: fecha, nivel en mayúscula y mensaje
  format: winston.format.combine(
    winston.format.timestamp({ format: 'YYYY-MM-DD HH:mm:ss' }),
    winston.format.printf(({ timestamp, level, message }) => `${timestamp} [${level.toUpperCase()}] ${message}`)
  ),
  // Dónde se escriben los logs: en la terminal y en el archivo logs/app.log
  transports: [
    new winston.transports.Console(),
    new winston.transports.File({ filename: 'logs/app.log' })
  ]
});

// Función para registrar un error con su stacktrace completo (nivel ERROR)
logger.fallo = (mensaje, error) => {
  // Si el error trae stack lo uso; si no, lo convierto a texto
  const detalle = error && error.stack ? error.stack : String(error);
  // Escribo el mensaje y debajo el stacktrace
  logger.error(mensaje + '\n' + detalle);
};

// Exporto el logger para usarlo en otros archivos
module.exports = logger;

===== src\controllers\reservaControlador.js =====
// Estados permitidos para una reserva
const ESTADOS = ['pendiente', 'confirmada', 'atendida', 'cancelada'];

// Toma solo los campos permitidos del cuerpo y les quita espacios
function limpiar(cuerpo) {
  const datos = {};
  ['clienteNombre', 'telefono', 'servicio', 'fechaHora', 'estado'].forEach((campo) => {
    if (typeof cuerpo[campo] === 'string') datos[campo] = cuerpo[campo].trim();
  });
  return datos;
}

// Devuelve un mensaje de error o null; completo = true exige los obligatorios (crear)
function validar(datos, completo) {
  if (completo) {
    for (const campo of ['clienteNombre', 'servicio', 'fechaHora']) {
      if (!datos[campo]) return 'Falta el campo obligatorio: ' + campo;
    }
  }
  for (const campo of ['clienteNombre', 'servicio', 'fechaHora']) {
    if (campo in datos && !datos[campo]) return 'El campo ' + campo + ' no puede estar vacío';
  }
  if (datos.estado && !ESTADOS.includes(datos.estado)) {
    return 'Estado no válido. Usa: ' + ESTADOS.join(', ');
  }
  return null;
}

// Inyección de dependencias: recibe el servicio de la base y el notificador desde afuera
function crearReservaControlador(servicio, notificador) {
  return {
    // POST /reservas: guarda en la base y luego envía a la API externa
    crear: async (req, res) => {
      try {
        const datos = limpiar(req.body || {});
        const error = validar(datos, true);
        if (error) return res.status(400).json({ error });
        // Paso 1: guardo en la base de datos
        const reserva = await servicio.crear(datos);
        // Paso 2: envío a la API externa; si falla, la reserva igual queda guardada
        let notificacion = 'no enviada';
        try {
          const externo = await notificador.notificar(reserva);
          notificacion = 'enviada (id externo ' + externo.id + ')';
        } catch (e) {
          console.error('No se pudo notificar a la API externa:', e.message);
        }
        res.status(201).json({ ...reserva, notificacion });
      } catch (error) {
        console.error('Error en POST /reservas:', error.message);
        res.status(500).json({ error: 'No se pudo crear la reserva' });
      }
    },

    // GET /reservas: todas
    listar: async (req, res) => {
      try { res.json(await servicio.listar()); }
      catch (error) {
        console.error('Error en GET /reservas:', error.message);
        res.status(500).json({ error: 'No se pudieron obtener las reservas' });
      }
    },

    // GET /reservas/:id: una por id (404 si no existe)
    obtener: async (req, res) => {
      try {
        const id = Number(req.params.id);
        if (!Number.isInteger(id)) return res.status(400).json({ error: 'El id debe ser un número' });
        const reserva = await servicio.obtener(id);
        if (!reserva) return res.status(404).json({ error: 'Reserva no encontrada' });
        res.json(reserva);
      } catch (error) {
        console.error('Error en GET /reservas/:id:', error.message);
        res.status(500).json({ error: 'No se pudo obtener la reserva' });
      }
    },

    // PUT /reservas/:id: modifica (404 si no existe)
    actualizar: async (req, res) => {
      try {
        const id = Number(req.params.id);
        if (!Number.isInteger(id)) return res.status(400).json({ error: 'El id debe ser un número' });
        const datos = limpiar(req.body || {});
        const error = validar(datos, false);
        if (error) return res.status(400).json({ error });
        const reserva = await servicio.actualizar(id, datos);
        if (!reserva) return res.status(404).json({ error: 'Reserva no encontrada' });
        res.json(reserva);
      } catch (error) {
        console.error('Error en PUT /reservas/:id:', error.message);
        res.status(500).json({ error: 'No se pudo actualizar la reserva' });
      }
    },

    // DELETE /reservas/:id: elimina y confirma (404 si no existe)
    eliminar: async (req, res) => {
      try {
        const id = Number(req.params.id);
        if (!Number.isInteger(id)) return res.status(400).json({ error: 'El id debe ser un número' });
        const reserva = await servicio.eliminar(id);
        if (!reserva) return res.status(404).json({ error: 'Reserva no encontrada' });
        res.json({ mensaje: 'Reserva eliminada', id });
      } catch (error) {
        console.error('Error en DELETE /reservas/:id:', error.message);
        res.status(500).json({ error: 'No se pudo eliminar la reserva' });
      }
    }
  };
}

module.exports = crearReservaControlador;

const URL_CAL = 'https://api.cal.com/v2';

// Función que arma los encabezados que Cal.com exige; recibe la versión de la API a usar
function encabezadosCal(version) {
  // Devuelvo un objeto con los dos encabezados
  return {
    // Authorization: aquí va mi clave. "Bearer" es la forma en que Cal.com espera recibirla,
    // y process.env.CAL_API_KEY es el valor guardado en el archivo .env
    Authorization: 'Bearer ' + process.env.CAL_API_KEY,
    // cal-api-version: le digo a Cal.com qué versión de su API quiero usar
    'cal-api-version': version
  };
}

// Función que trae la lista de clientes de prueba
// async porque pedir datos por internet tarda y la función tiene que esperar
async function obtenerClientes() {
  // await = espera a que JSONPlaceholder responda; pido /users y no mando clave porque no la pide
  const respuesta = await axios.get(`${URL_BASE}/users`);
  // respuesta.data es el JSON con los datos; eso es lo que devuelvo
  return respuesta.data;
}

// Función que trae los servicios de la barbería (los tipos de evento de Cal.com)
async function obtenerServicios() {
  // Pido /event-types a Cal.com y le mando los encabezados con mi clave y la versión 2024-06-14
  const respuesta = await axios.get(`${URL_CAL}/event-types`, { headers: encabezadosCal('2024-06-14') });
  // Cal.com mete la lista dentro de un campo llamado "data"; la guardo en esta variable
  const datos = respuesta.data.data;
  // Si lo que llegó es una lista la devuelvo; si no, devuelvo una lista vacía para no romper la página
  return Array.isArray(datos) ? datos : [];
}

// Función que trae las citas apartadas (las reservas de Cal.com)
async function obtenerCitas() {
  // Pido /bookings a Cal.com con mi clave; esta ruta pide la versión 2026-05-01
  const respuesta = await axios.get(`${URL_CAL}/bookings`, { headers: encabezadosCal('2026-05-01') });
  // La lista de reservas también viene dentro del campo "data"
  const datos = respuesta.data.data;
  // Devuelvo la lista, o una vacía si no llegó una lista
  return Array.isArray(datos) ? datos : [];
}

// Exporto las tres funciones para usarlas en otros archivos (el controlador y las pruebas)
module.exports = { obtenerClientes, obtenerServicios, obtenerCitas };

// Conexión a la base de datos y descripción de la tabla citas
const AppDataSource = require('../database/data-source');
const Cita = require('../entities/Cita');

// Repositorio: la herramienta del ORM para leer y guardar filas
function repo() { return AppDataSource.getRepository(Cita); }

// READ (todos)
async function listar() { return repo().find({ order: { id: 'ASC' } }); }

// READ (uno por id); devuelve null si no existe
async function obtener(id) { return repo().findOneBy({ id }); }

// CREATE: arma la fila y la guarda; devuelve la fila con su id
async function crear(datos) { return repo().save(repo().create(datos)); }

// UPDATE: busca, cambia solo lo recibido y guarda; null si no existe
async function actualizar(id, datos) {
  const cita = await repo().findOneBy({ id });
  if (!cita) return null;
  repo().merge(cita, datos);
  return repo().save(cita);
}

// DELETE: busca y borra; null si no existe
async function eliminar(id) {
  const cita = await repo().findOneBy({ id });
  if (!cita) return null;
  const copia = { ...cita };
  await repo().remove(cita);
  return copia;
}

module.exports = { listar, obtener, crear, actualizar, eliminar };

// Importo la conexión a la base de datos (la misma que uso en app.js)
const AppDataSource = require('../database/data-source');
// Importo mi servicio de APIs, que ya sabe pedir los datos a Cal.com y JSONPlaceholder
const apiService = require('./apiService');

// Esta función traduce el estado que usa Cal.com al estado que usa mi tabla de citas
function traducirEstado(estado) {
  // Si Cal.com dice "accepted", en mi base queda como "confirmada"
  if (estado === 'accepted') return 'confirmada';
  // Si Cal.com dice "cancelled", en mi base queda como "cancelada"
  if (estado === 'cancelled') return 'cancelada';
  // Cualquier otro estado lo guardo como "pendiente"
  return 'pendiente';
}

// Esta función trae los datos de las APIs y los guarda en la base de datos con el ORM
async function sincronizar() {
  // Objeto donde voy contando cuántos registros nuevos se guardaron de cada tipo
  const resumen = { clientes: 0, servicios: 0, citas: 0 };

  // ===== 1. CLIENTES (JSONPlaceholder -> tabla Cliente) =====
  try {
    // Pido al ORM el repositorio de Cliente (es el que me deja guardar y buscar clientes)
    const repo = AppDataSource.getRepository('Cliente');
    // Pido a la API externa la lista de clientes
    const clientesApi = await apiService.obtenerClientes();
    // Recorro uno por uno los clientes que trajo la API
    for (const c of clientesApi) {
      // Busco en mi base si ya existe un cliente con ese nombre (para no repetirlo)
      const existe = await repo.findOneBy({ nombre: c.name });
      // Si no existe, lo guardo
      if (!existe) {
        // create arma el cliente con mis columnas y save lo guarda en la base de datos
        await repo.save(repo.create({ nombre: c.name, telefono: c.phone }));
        // Sumo 1 al contador de clientes guardados
        resumen.clientes++;
      }
    }
  // Si algo falla (por ejemplo la API caída), muestro el error en la terminal y sigo con lo demás
  } catch (e) { console.error('Error sincronizando clientes:', e.message); }

  // ===== 2. SERVICIOS (Cal.com -> tabla Servicio) =====
  try {
    // Pido al ORM el repositorio de Servicio
    const repo = AppDataSource.getRepository('Servicio');
    // Pido a Cal.com la lista de servicios (tipos de evento)
    const serviciosApi = await apiService.obtenerServicios();
    // Recorro uno por uno los servicios que trajo la API
    for (const s of serviciosApi) {
      // Cal.com guarda el nombre en "title" (o en "name"); tomo el que exista
      const nombre = s.title || s.name;
      // Si el servicio no trae nombre, lo salto y paso al siguiente
      if (!nombre) continue;
      // Busco en mi base si ya existe un servicio con ese nombre
      const existe = await repo.findOneBy({ nombre });
      // Si no existe, lo guardo
      if (!existe) {
        // Cal.com no maneja precio en este endpoint, lo dejo en 0 para editarlo después
        await repo.save(repo.create({ nombre, precio: 0, activo: true }));
        // Sumo 1 al contador de servicios guardados
        resumen.servicios++;
      }
    }
  // Si algo falla, muestro el error en la terminal y sigo con lo demás
  } catch (e) { console.error('Error sincronizando servicios:', e.message); }

  // ===== 3. CITAS (Cal.com -> tabla Cita) =====
  try {
    // Pido al ORM el repositorio de Cita (ahí guardo las reservas)
    const repo = AppDataSource.getRepository('Cita');
    // Pido a Cal.com la lista de citas
    const citasApi = await apiService.obtenerCitas();
    // Recorro una por una las citas que trajo la API
    for (const c of citasApi) {
      // El nombre del cliente viene dentro de "attendees"; si no hay, pongo un texto por defecto
      const clienteNombre = c.attendees && c.attendees[0] ? c.attendees[0].name : 'Cliente sin nombre';
      // El servicio es el título de la cita; si no trae título, pongo un texto por defecto
      const servicio = c.title || 'Cita de Cal.com';
      // La fecha y hora de inicio de la cita viene en "start"
      const fechaHora = c.start;
      // Busco si ya guardé esa misma cita (mismo cliente, servicio y fecha) para no repetirla
      const existe = await repo.findOneBy({ clienteNombre, servicio, fechaHora });
      // Si no existe, la guardo
      if (!existe) {
        // create arma la cita con mis columnas y save la guarda en la base de datos
        await repo.save(repo.create({
          clienteNombre,                         // nombre del cliente
          telefono: '',                          // Cal.com no manda teléfono, lo dejo vacío
          servicio,                              // nombre del servicio
          fechaHora,                             // fecha y hora de la cita
          estado: traducirEstado(c.status)       // estado traducido a los que usa mi proyecto
        }));
        // Sumo 1 al contador de citas guardadas
        resumen.citas++;
      }
    }
  // Si algo falla, muestro el error en la terminal
  } catch (e) { console.error('Error sincronizando citas:', e.message); }

  // Devuelvo el resumen con cuántos registros nuevos se guardaron de cada tipo
  return resumen;
}

function esBarbero() { return claveBarbero !== ''; }
// Arma los encabezados; si es el barbero, manda su clave para que el servidor lo reconozca
function cabeceras() {
  const h = { 'Content-Type': 'application/json' };
  if (esBarbero()) h['x-clave-barbero'] = claveBarbero;
  return h;
}

// Muestra u oculta lo del barbero y cambia los textos de la barra
function mostrarModo() {
  document.body.classList.toggle('modo-barbero', esBarbero());
  document.getElementById('estado-barbero').textContent = esBarbero() ? 'Modo barbero' : 'Vista de cliente';
  document.getElementById('btn-barbero').textContent = esBarbero() ? 'Salir del modo barbero' : 'Soy el barbero';
}

// Botón "Soy el barbero": pide la clave y la comprueba con el servidor; si ya estaba dentro, sale
async function alternarBarbero() {
  if (esBarbero()) {
    claveBarbero = '';
    sessionStorage.removeItem('claveBarbero');
  } else {
    const clave = prompt('Clave del barbero:');
    if (!clave) return;
    const res = await fetch('/login-barbero', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ clave }) });
    if (!res.ok) { alert('Clave incorrecta'); return; }
    claveBarbero = clave;
    sessionStorage.setItem('claveBarbero', clave);
  }
  mostrarModo();
  cargarTodo();
}

// Pide datos a mi API y devuelve el JSON
async function pedir(ruta) {
  const respuesta = await fetch(ruta);
  if (!respuesta.ok) throw new Error('El servidor respondió con error');
  return respuesta.json();
}

// Arma una tarjeta con título, líneas de texto, etiqueta opcional y botón opcional
function crearTarjeta(titulo, lineas, etiqueta, boton) {
  const tarjeta = document.createElement('article');
  tarjeta.className = 'tarjeta';
  const h3 = document.createElement('h3');
  h3.textContent = titulo;
  tarjeta.appendChild(h3);
  lineas.forEach((texto) => {
    const p = document.createElement('p');
    p.textContent = texto;
    tarjeta.appendChild(p);
  });
  if (etiqueta) {
    const span = document.createElement('span');
    span.className = 'etiqueta ' + etiqueta.clase;
    span.textContent = etiqueta.texto;
    tarjeta.appendChild(span);
  }
  if (boton) {
    const a = document.createElement('a');
    a.className = 'boton-reservar';
    a.textContent = boton.texto;
    a.href = boton.url;
    a.target = '_blank';
    a.rel = 'noopener';
    tarjeta.appendChild(a);
  }
  return tarjeta;
}

// Crea un botón pequeño (editar o borrar)
function crearBoton(texto, clase, accion) {
  const b = document.createElement('button');
  b.textContent = texto;
  b.className = clase;
  b.onclick = accion;
  return b;
}

// Convierte una fecha en texto legible, en hora de Colombia
function formatearFecha(texto) {
  const fecha = new Date(texto);
  if (isNaN(fecha)) return 'Fecha no disponible';
  return fecha.toLocaleString('es-CO', { dateStyle: 'medium', timeStyle: 'short', timeZone: 'America/Bogota' });
}

// Si el servidor rechazó la acción (401), aviso que es solo del barbero
function avisarSiRechazo(res, mensaje) {
  if (res.status === 401) alert('Solo el barbero puede hacer esto');
  else if (!res.ok) alert(mensaje);
  return res.ok;
}

// ===== CLIENTES: todos agregan y editan; borrar solo el barbero =====
async function cargarClientes() {
  const cont = document.getElementById('clientes');
  try {
    const [clientesApi, clientesLocales] = await Promise.all([
      pedir('/clientes').catch(() => []),
      pedir('/clientes-locales').catch(() => [])
    ]);
    const total = [...clientesApi, ...clientesLocales];
    document.getElementById('fuente-clientes').textContent =
      'Fuente: API externa + Base de datos SQLite · ' + total.length + ' registros recibidos';
    cont.textContent = '';
    if (total.length === 0) { cont.textContent = 'Todavía no hay clientes registrados'; return; }

    clientesApi.forEach((c) => cont.appendChild(crearTarjeta(c.name, [c.phone, c.email])));

    clientesLocales.forEach((c) => {
      const id = c.id;
      const tarjeta = crearTarjeta(c.nombre || 'Cliente Local', [c.telefono ? 'Teléfono: ' + c.telefono : 'Sin teléfono'], { texto: 'BD Local', clase: 'atendida' });

      // Editar: lo puede hacer cualquiera
      tarjeta.appendChild(crearBoton('Editar', 'btn-accion', async () => {
        const nuevoNombre = prompt('Nuevo nombre:', c.nombre);
        const nuevoTelefono = prompt('Nuevo teléfono:', c.telefono);
        if (nuevoNombre && id) {
          const res = await fetch('/clientes-locales/' + id, { method: 'PUT', headers: cabeceras(), body: JSON.stringify({ nombre: nuevoNombre, telefono: nuevoTelefono }) });
          if (avisarSiRechazo(res, 'No se pudo actualizar el cliente')) cargarClientes();
        }
      }));

      // Borrar: el botón solo existe para el barbero
      if (esBarbero()) {
        tarjeta.appendChild(crearBoton('Borrar', 'btn-accion btn-eliminar', async () => {
          if (confirm('¿Seguro que deseas eliminar este cliente?')) {
            const res = await fetch('/clientes-locales/' + id, { method: 'DELETE', headers: cabeceras() });
            if (avisarSiRechazo(res, 'No se pudo eliminar el cliente')) cargarClientes();
          }
        }));
      }
      cont.appendChild(tarjeta);
    });
  } catch (error) {
    cont.textContent = 'No se pudieron cargar los clientes';
  }
}

async function agregarCliente(e) {
  e.preventDefault();
  const nombre = document.getElementById('nombreCliente').value;
  const telefono = document.getElementById('telefonoCliente').value;
  const res = await fetch('/clientes-locales', { method: 'POST', headers: cabeceras(), body: JSON.stringify({ nombre, telefono }) });
  if (avisarSiRechazo(res, 'Error al guardar el cliente')) {
    document.getElementById('form-cliente').reset();
    cargarClientes();
  }
}

// ===== SERVICIOS: agregar, editar y borrar solo el barbero =====
async function cargarServicios() {
  const cont = document.getElementById('servicios');
  const selectCita = document.getElementById('servicioCita');
  if (selectCita) selectCita.innerHTML = '<option value="">Seleccione un servicio...</option>';

  try {
    const [serviciosCal, serviciosLocales] = await Promise.all([
      pedir('/servicios').catch(() => []),
      pedir('/servicios-locales').catch(() => [])
    ]);
    const total = [...serviciosCal, ...serviciosLocales];
    document.getElementById('fuente-servicios').textContent =
      'Fuente: API Cal.com + Base de datos SQLite · ' + total.length + ' registros recibidos';
    cont.textContent = '';
    if (total.length === 0) { cont.textContent = 'Todavía no hay servicios creados'; return; }

    // Servicios de Cal.com (solo se ven)
    serviciosCal.filter((s) => !s.hidden).forEach((s) => {
      const nombre = s.title || s.name || 'Servicio';
      const minutos = s.lengthInMinutes || s.length;
      const lineas = [];
      if (s.description) lineas.push(s.description);
      if (minutos) lineas.push('Duración: ' + minutos + ' minutos');
      const url = s.bookingUrl || ('https://cal.com/' + USUARIO_CAL + '/' + encodeURIComponent(s.slug));
      cont.appendChild(crearTarjeta(nombre, lineas, null, { texto: 'Pedir cita', url }));
      if (selectCita) {
        const option = document.createElement('option');
        option.value = nombre;
        option.textContent = nombre;
        selectCita.appendChild(option);
      }
    });

    // Servicios de mi base de datos
    serviciosLocales.forEach((s) => {
      const id = s.id;
      const lineas = [];
      if (s.precio !== undefined && s.precio !== null) lineas.push('Precio: $' + Number(s.precio).toLocaleString('es-CO'));
      const tarjeta = crearTarjeta(s.nombre || 'Servicio Local', lineas, { texto: 'BD Local', clase: 'atendida' }, { texto: 'Pedir cita', url: 'https://cal.com/' + USUARIO_CAL });

      // Editar y borrar: los botones solo existen para el barbero
      if (esBarbero()) {
        tarjeta.appendChild(crearBoton('Editar', 'btn-accion', async () => {
          const nuevoNombre = prompt('Nuevo nombre del servicio:', s.nombre);
          const nuevoPrecioStr = prompt('Nuevo precio ($):', s.precio);
          if (nuevoNombre !== null && id) {
            let nuevoPrecio = Number(nuevoPrecioStr);
            if (nuevoPrecio < 1000) nuevoPrecio = nuevoPrecio * 1000;
            const res = await fetch('/servicios-locales/' + id, { method: 'PUT', headers: cabeceras(), body: JSON.stringify({ nombre: nuevoNombre, precio: isNaN(nuevoPrecio) ? 0 : nuevoPrecio, activo: true }) });
            if (avisarSiRechazo(res, 'No se pudo actualizar el servicio')) cargarServicios();
          }
        }));
        tarjeta.appendChild(crearBoton('Borrar', 'btn-accion btn-eliminar', async () => {
          if (confirm('¿Seguro que deseas eliminar este servicio?')) {
            const res = await fetch('/servicios-locales/' + id, { method: 'DELETE', headers: cabeceras() });
            if (avisarSiRechazo(res, 'No se pudo eliminar el servicio')) cargarServicios();
          }
        }));
      }
      cont.appendChild(tarjeta);

      if (selectCita) {
        const option = document.createElement('option');
        option.value = s.nombre;
        option.textContent = s.nombre + (s.precio !== undefined ? ' ($' + Number(s.precio).toLocaleString('es-CO') + ')' : '');
        selectCita.appendChild(option);
      }
    });
  } catch (error) {
    cont.textContent = 'No se pudieron cargar los servicios';
  }
}

async function agregarServicio(e) {
  e.preventDefault();
  const nombre = document.getElementById('nombreServicio').value;
  let precio = parseFloat(document.getElementById('precioServicio').value);
  if (precio < 1000) precio = precio * 1000;
  const res = await fetch('/servicios-locales', { method: 'POST', headers: cabeceras(), body: JSON.stringify({ nombre, precio, activo: true }) });
  if (avisarSiRechazo(res, 'Error al guardar el servicio')) {
    document.getElementById('form-servicio').reset();
    cargarServicios();
  }
}

===== README.md =====
# Barber Club - API REST, Base de Datos, Observabilidad y Seguridad

Proyecto de aplicación web para la gestión de la barbería MVP Barber Club. El sistema combina el consumo de APIs externas (Cal.com y JSONPlaceholder) con persistencia de datos local mediante **TypeORM** y **SQLite**, con un flujo CRUD completo. Incluye registro de logs, un endpoint de salud, medidas de seguridad y pruebas de integración.

---

## Tecnologías e infraestructura
- **Backend:** Node.js con Express.
- **Base de datos local:** SQLite administrada con **TypeORM**. Las tablas las crea el ORM a partir de las entidades (synchronize: true), sin SQL manual.
- **Frontend:** interfaz web en HTML5, CSS3 y JavaScript, con la API fetch para hablar con el servidor.
- **Integraciones externas:** API de Cal.com (servicios y citas) y JSONPlaceholder (clientes de prueba y envío de reservas).
- **Logs:** winston.
- **Pruebas:** Jest y Supertest.

---

## Cómo ejecutarlo
1. Instalar las librerías: `npm install`
2. Crear un archivo `.env` en la raíz con estas dos variables:
   ```
   CAL_API_KEY=tu_clave_de_cal
   CLAVE_BARBERO=la_clave_del_barbero
   ```
3. Encender el servidor: `npm start` (o `node src/app.js`)
4. Abrir http://localhost:3000

Al arrancar, el servidor copia a la base de datos los datos de las APIs (sin repetir los que ya existen).

## Cómo ejecutar las pruebas
`npm test`

Las pruebas de integración usan una base de datos en memoria, así que no tocan `barberia.sqlite`.

---

## Endpoints (API REST)

### 1. Salud del servidor
- `GET /health`: responde `{ "status": "ok", "timestamp": "..." }` para comprobar que el servidor está vivo.

### 2. Servicios externos (Reto 1)
- `GET /servicios`: lista de servicios agendables desde la API de Cal.com.
- `GET /citas`: citas programadas desde Cal.com.
- `GET /clientes`: clientes de prueba obtenidos desde JSONPlaceholder.

### 3. Persistencia y CRUD local (Reto 2)
- **CRUD de reservas (entidad Cita):**
  - `POST /reservas`: valida los datos, guarda en SQLite y envía la reserva a una API externa. Responde 201 con la reserva y su id.
  - `GET /reservas`: lista todas las reservas.
  - `GET /reservas/:id`: busca una reserva por su id (404 si no existe).
  - `PUT /reservas/:id`: modifica una reserva y la devuelve (404 si no existe). **Solo el barbero.**
  - `DELETE /reservas/:id`: elimina una reserva y confirma (404 si no existe). **Solo el barbero.**
- **Servicios locales (`/servicios-locales`):** GET es de todos; POST, PUT y DELETE son solo del barbero.
- **Clientes locales (`/clientes-locales`):** GET, POST y PUT validan los datos de entrada; DELETE es solo del barbero.
- `POST /sincronizar`: trae los datos de las APIs y los guarda en la base con el ORM. **Solo el barbero.**
- `DELETE /datos`: borra los datos locales. **Solo el barbero.**
- `POST /login-barbero`: comprueba la clave del barbero.

Las acciones del barbero exigen el encabezado `x-clave-barbero` con la clave guardada en el `.env`.

---

## Envío a una API externa
Al crear una reserva (`POST /reservas`), la aplicación la guarda en la base de datos con TypeORM y después la envía con POST a JSONPlaceholder (https://jsonplaceholder.typicode.com/posts). La respuesta incluye el estado del envío. Si la API externa falla, la reserva igual queda guardada.

## Sincronización de la API a la base de datos
El servicio `src/services/sincronizarService.js` trae los clientes (JSONPlaceholder), los servicios y las citas (Cal.com) y los guarda en la base de datos usando solo el ORM (`create` y `save`). Antes de guardar revisa si el registro ya existe, para no repetirlo. Se ejecuta al arrancar el servidor y también con `POST /sincronizar`.

---

## Observabilidad (Reto 3, parte 1)
- **Logs:** el archivo `src/logger.js` configura winston. Un middleware registra cada petición con método, ruta, código HTTP y tiempo (por ejemplo `GET /health 200 4ms`).
- **Errores:** se registran con nivel `[ERROR]` y el stacktrace completo, en la terminal y en el archivo `logs/app.log`.
- **Endpoint `/health`:** descrito arriba.

## Seguridad (Reto 3, parte 2)
| Vulnerabilidad | Riesgo | Corrección |
|---|---|---|
| Sin validación de entrada | Se guardaban nombres vacíos, teléfonos inválidos y un `id` forzado | `limpiarCliente` y `validarCliente`: solo pasan `nombre` y `telefono`, con reglas de largo y formato. Responde 400 |
| Endpoints sin autenticación | Cualquiera podía modificar reservas | `PUT /reservas/:id` exige la clave del barbero. Responde 401 |
| Errores técnicos expuestos | El stacktrace mostraba rutas internas, y el encabezado `X-Powered-By` revelaba Express | Manejador de errores con mensajes genéricos (el detalle va al log) y `app.disable('x-powered-by')` |

Además, las claves están en el archivo `.env`, que no se sube al repositorio.

## Pruebas de integración (Reto 3, parte 3)
En `tests/integracion.test.js`, con Jest y Supertest, sobre una base de datos en memoria:
1. **Create:** `POST /reservas` responde 201 y el dato queda en la base.
2. **Read:** se guarda una reserva y `GET /reservas/:id` devuelve exactamente ese dato.
3. **Delete:** se crea y se borra una reserva, y el `GET` siguiente responde 404.

En `tests/pruebas.test.js` están las pruebas del Reto 1, con servicios simulados (mocks).

---

## Inyección de dependencias
Los controladores no crean sus servicios: los reciben desde afuera. Así, en las pruebas se les pasa un servicio simulado (mock) y se prueba sin internet ni base de datos.
